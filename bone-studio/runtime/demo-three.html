Dùng trong game three.js

import * as THREE from "three";
import { loadSkeleton } from "./xuong-player.js";
import { XuongMesh } from "./xuong-three.js";
const data = await loadSkeleton("pharaoh.xuong.json");
const m = new XuongMesh(THREE, data,
  { pixelsPerUnit: 100 });
m.play("walk", { mix: 0.2 });
scene.add(m);      // m là THREE.Object3D
// mỗi khung:
for (const e of m.update(dt)) onEvent(e.name);
// m.bounds() → hộp bao (đơn vị three)
