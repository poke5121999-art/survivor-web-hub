[script lab_enter]
if met_oak > end
set met_oak
face oak down
say@Prof._Oak Ah, {player}! There you are.

[script oak]
if starter_chosen > after
say Go ahead, {player}. Choose one of the Pokémon on the machine!
end
:after
say {player}, Route 1 is north of Pallet Town.\nA young trainer named Joey is waiting there.
say Why not challenge him and see what your Pokémon can do?

[script lab_aide]
say Prof. Oak's research is about Pokémon habitats.\nHe says Pokémon and people should live side by side.

[script gary]
if starter_chosen > after
say Heh, I don't need to be greedy like you.\nGo ahead and choose, {player}!
end
:after
say My Pokémon looks a lot tougher than yours.

[script starter_machine]
if starter_chosen > done
:pick
choose Which Pokémon will you choose? | Bulbasaur > b | Charmander > c | Squirtle > s | Not yet > end
:b
showmon 1
cry 1
choose So you want the Grass-type Pokémon, Bulbasaur? | Yes > take_b | No > back
:c
showmon 4
cry 4
choose So you want the Fire-type Pokémon, Charmander? | Yes > take_c | No > back
:s
showmon 7
cry 7
choose So you want the Water-type Pokémon, Squirtle? | Yes > take_s | No > back
:back
hidemon
goto pick
:take_b
hidemon
givemon 1 5
set starter_1
goto rival
:take_c
hidemon
givemon 4 5
set starter_4
goto rival
:take_s
hidemon
givemon 7 5
set starter_7
:rival
set starter_chosen
say@Gary Then I'll take this one!

face gary left
say@Gary Wait, {player}! Let's check out our Pokémon!\nCome on, I'll take you on!
if starter_1 > vs_b
if starter_4 > vs_c
battle name=Gary team=1:5 music=trainer_battle canlose win=won lose=lost
goto won
:vs_b
battle name=Gary team=4:5 music=trainer_battle canlose win=won lose=lost
goto won
:vs_c
battle name=Gary team=7:5 music=trainer_battle canlose win=won lose=lost
:won
say@Gary What? Unbelievable!\nI picked the wrong Pokémon!
goto after_battle
:lost
say@Gary Yeah! Am I great or what?
:after_battle
heal
questdone first_battle
say@Gary Okay! I'll make my Pokémon fight to toughen it up!\nSmell you later, {player}!
hide gary_left
quest trainer_route_1
say@Prof._Oak {player}, a trainer named Joey is waiting on Route 1.\nGo challenge him!
end
:done
say It's a machine that holds Poké Balls.
